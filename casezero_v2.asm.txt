; ============================================================================
; main - game loop
; ============================================================================
main PROC
    call Randomize               ; seed the random number generator once

StartGame:
    call ResetGame               ; clears state and picks a random case
    call Clrscr
    call ShowIntro

MenuLoop:
    call NewScreen
    mov  edx, OFFSET msgPromptMain
    call WriteString
    mov  esi, OFFSET mainMenuTbl
    mov  ecx, MAIN_ITEMS
    mov  ebx, MENU_ROW
    call ArrowMenu               ; EAX = 0..6, or -1 if ESC
    cmp  eax, 0
    je   DoClue
    cmp  eax, 1
    je   DoSuspect
    cmp  eax, 2
    je   DoConfront
    cmp  eax, 3
    je   DoPuzzle
    cmp  eax, 4
    je   DoEvidence
    cmp  eax, 5
    je   DoStatus
    cmp  eax, 6
    je   DoAccuse
    jmp  MenuLoop                ; ESC: just redraw

DoClue:
    call InspectClue
    jmp  AfterAction
DoSuspect:
    call QuestionSuspect
    jmp  AfterAction
DoConfront:
    call Confront
    jmp  AfterAction
DoPuzzle:
    call SolvePuzzle
    jmp  AfterAction
DoEvidence:
    call ShowEvidence
    jmp  AfterAction
DoStatus:
    call ShowStatus
    jmp  AfterAction
DoAccuse:
    call CheckAccusation         ; EAX = 1 if the case ended, 0 otherwise
    cmp  eax, 1
    je   GameOver

AfterAction:
    call Crlf
    call WaitMsg                 ; "Press any key to continue..."
    jmp  MenuLoop

GameOver:
    mov  edx, OFFSET msgReplayQ
    mov  ebx, OFFSET msgReplayT
    call MsgBoxAsk               ; Yes/No pop-up window, EAX = 6 for Yes
    cmp  eax, ANSWER_YES
    je   StartGame
    mov  edx, OFFSET msgBye
    call WriteString
    exit
main ENDP

; ============================================================================
; SetupCase - picks a random case and points the cur... variables at its data
; ============================================================================
SetupCase PROC
    push eax
    push ebx

    mov  eax, NUM_CASES
    call RandomRange             ; EAX = 0 .. NUM_CASES-1
    mov  curCase, eax
    mov  ebx, eax

    mov  eax, caseClueTbl[ebx*4]
    mov  curClueTable, eax
    mov  eax, caseLabelTbl[ebx*4]
    mov  curLabelTable, eax
    mov  eax, caseEvidTbl[ebx*4]
    mov  curEvidTable, eax
    mov  eax, caseConfTbl[ebx*4]
    mov  curConfrontTable, eax
    mov  eax, caseSolveTbl[ebx*4]
    mov  curSolveText, eax
    mov  eax, caseCulprit[ebx*4]
    mov  culprit, eax

    pop  ebx
    pop  eax
    ret
SetupCase ENDP

; ============================================================================
; ResetGame - clears all arrays and the score (STOSB), then picks a new case
; ============================================================================
ResetGame PROC
    push eax
    push ecx
    push edi

    mov  score, 0
    cld                          ; string ops go forward
    mov  al, 0                   ; value to store

    mov  edi, OFFSET clueFound
    mov  ecx, NUM_CLUES
    rep  stosb

    mov  edi, OFFSET suspectAsked
    mov  ecx, NUM_SUSPECTS
    rep  stosb

    mov  edi, OFFSET puzzleDone
    mov  ecx, NUM_PUZZLES
    rep  stosb

    mov  edi, OFFSET confrontDone
    mov  ecx, NUM_SUSPECTS*NUM_CLUES
    rep  stosb

    call SetupCase

    pop  edi
    pop  ecx
    pop  eax
    ret
ResetGame ENDP

; ============================================================================
; ShowIntro - title screen and case description
; ============================================================================
ShowIntro PROC
    push eax
    push edx

    mov  eax, yellow + (black * 16)
    call SetTextColor
    mov  edx, OFFSET titleBar
    call WriteString

    mov  eax, white + (black * 16)
    call SetTextColor
    mov  edx, OFFSET introText
    call WriteString
    call WaitMsg

    pop  edx
    pop  eax
    ret
ShowIntro ENDP

; ============================================================================
; NewScreen - clears the console and draws the status header (rows 0-1)
;             the cursor is left on row 3, ready for a prompt line
; ============================================================================
NewScreen PROC
    call Clrscr
    call ShowHeader
    ret
NewScreen ENDP

; ============================================================================
; ShowHeader - coloured title bar + live score / progress line
; ============================================================================
ShowHeader PROC
    push eax
    push ecx
    push edx
    push esi

    mov  dh, 0                   ; row 0, column 0
    mov  dl, 0
    call Gotoxy

    mov  eax, yellow + (blue * 16)
    call SetTextColor
    mov  edx, OFFSET hdrBar
    call WriteString
    mov  eax, lightGray + (black * 16)
    call SetTextColor
    call Crlf

    mov  edx, OFFSET hdrScore
    call WriteString
    mov  eax, score
    call WriteDec

    mov  edx, OFFSET hdrClues
    call WriteString
    mov  esi, OFFSET clueFound
    mov  ecx, NUM_CLUES
    call CountFlags
    call WriteDec
    mov  edx, OFFSET msgSlash
    call WriteString
    mov  eax, NUM_CLUES
    call WriteDec

    mov  edx, OFFSET hdrNotes
    call WriteString
    mov  esi, OFFSET suspectAsked
    mov  ecx, NUM_SUSPECTS
    call CountFlags
    call WriteDec
    mov  edx, OFFSET msgSlash
    call WriteString
    mov  eax, NUM_SUSPECTS
    call WriteDec

    mov  edx, OFFSET hdrProofs
    call WriteString
    mov  esi, OFFSET confrontDone
    mov  ecx, NUM_SUSPECTS*NUM_CLUES
    call CountFlags
    call WriteDec
    mov  edx, OFFSET msgSlash
    call WriteString
    mov  eax, NUM_PROOFS
    call WriteDec

    mov  eax, white + (black * 16)
    call SetTextColor
    mov  dh, 3                   ; prompt line goes on row 3
    mov  dl, 0
    call Gotoxy

    pop  esi
    pop  edx
    pop  ecx
    pop  eax
    ret
ShowHeader ENDP

; ============================================================================
; DrawMenu - redraws every item of the open menu; the selected one is
;            shown with a cyan highlight bar
; ============================================================================
DrawMenu PROC
    push eax
    push ebx
    push ecx
    push edx
    push esi

    mov  esi, menuTbl
    mov  ebx, 0                  ; item number
DM_Loop:
    mov  edx, menuRow
    add  edx, ebx
    mov  dh, dl                  ; DH = row  (menuRow + item)
    mov  dl, 2                   ; DL = column
    call Gotoxy

    cmp  ebx, menuSel
    jne  DM_Normal
    mov  eax, black + (lightCyan * 16)
    call SetTextColor
    mov  al, '>'
    call WriteChar
    mov  al, ' '
    call WriteChar
    jmp  DM_Text
DM_Normal:
    mov  eax, white + (black * 16)
    call SetTextColor
    mov  al, ' '
    call WriteChar
    mov  al, ' '
    call WriteChar
DM_Text:
    mov  edx, [esi+ebx*4]
    call WriteString
    inc  ebx
    cmp  ebx, menuCnt
    jb   DM_Loop

    mov  eax, white + (black * 16)
    call SetTextColor

    pop  esi
    pop  edx
    pop  ecx
    pop  ebx
    pop  eax
    ret
DrawMenu ENDP

; ============================================================================
; ArrowMenu - interactive menu driven by the arrow keys
;   IN : ESI = table of string pointers, ECX = number of items,
;        EBX = screen row of the first item
;   OUT: EAX = chosen item (0-based), or -1 if the player pressed ESC
;   The cursor is left two rows below the last item.
;   (ReadKey overwrites EBX, so all state lives in the menu... variables.)
; ============================================================================
ArrowMenu PROC
    push ebx
    push ecx
    push edx
    push esi

    mov  menuTbl, esi
    mov  menuCnt, ecx
    mov  menuRow, ebx
    mov  menuSel, 0

AM_Draw:
    call DrawMenu
AM_Wait:
    mov  eax, 30
    call Delay                   ; give the keyboard queue time to fill
    call ReadKey                 ; ZF=1 -> no key.  AL = ASCII, DX = virtual key
    jz   AM_Wait

    cmp  dx, K_UP
    je   AM_Up
    cmp  al, 'w'
    je   AM_Up
    cmp  al, 'W'
    je   AM_Up
    cmp  dx, K_DOWN
    je   AM_Down
    cmp  al, 's'
    je   AM_Down
    cmp  al, 'S'
    je   AM_Down
    cmp  dx, K_ENTER
    je   AM_Enter
    cmp  al, 0Dh                 ; fallback if DX is not filled in
    je   AM_Enter
    cmp  dx, K_ESC
    je   AM_Esc
    cmp  al, 1Bh
    je   AM_Esc
    jmp  AM_Wait

AM_Up:
    mov  eax, menuSel
    cmp  eax, 0
    jne  AM_UpDec
    mov  eax, menuCnt            ; wrap from the top to the bottom
AM_UpDec:
    dec  eax
    mov  menuSel, eax
    jmp  AM_Draw

AM_Down:
    mov  eax, menuSel
    inc  eax
    cmp  eax, menuCnt
    jb   AM_DownOk
    mov  eax, 0                  ; wrap from the bottom to the top
AM_DownOk:
    mov  menuSel, eax
    jmp  AM_Draw

AM_Enter:
    mov  eax, menuSel
    jmp  AM_Finish
AM_Esc:
    mov  eax, -1

AM_Finish:
    mov  edx, menuRow            ; put the cursor below the menu
    add  edx, menuCnt
    inc  edx
    mov  dh, dl                  ; DH = row
    mov  dl, 0                   ; DL = column
    call Gotoxy

    pop  esi
    pop  edx
    pop  ecx
    pop  ebx
    ret
ArrowMenu ENDP

; ============================================================================
; TypeString - prints a string one letter at a time (typewriter effect)
;   IN : EDX = string address
; ============================================================================
TypeString PROC
    push eax
    push esi

    mov  esi, edx
TS_Loop:
    mov  al, [esi]
    cmp  al, 0
    je   TS_Done
    call WriteChar
    cmp  BYTE PTR [esi], 0dh     ; no pause on line breaks
    je   TS_Next
    cmp  BYTE PTR [esi], 0ah
    je   TS_Next
    mov  eax, TYPE_DELAY
    call Delay
TS_Next:
    inc  esi
    jmp  TS_Loop
TS_Done:
    pop  esi
    pop  eax
    ret
TypeString ENDP

; ============================================================================
; CountFlags - counts non-zero bytes in a byte array
;   IN : ESI = array address, ECX = array length (must be > 0)
;   OUT: EAX = number of non-zero entries
; ============================================================================
CountFlags PROC
    push esi
    push ecx
    mov  eax, 0
CF_Loop:
    cmp  BYTE PTR [esi], 0
    je   CF_Next
    inc  eax
CF_Next:
    inc  esi
    loop CF_Loop
    pop  ecx
    pop  esi
    ret
CountFlags ENDP

; ============================================================================
; InspectClue - choose a LOCATION to search; the clue there is logged

IC_Already:

    call WriteString

IC_Done:

    pop  edx
    pop  ecx
    pop  ebx
    pop  eax
    ret
InspectClue ENDP

; ============================================================================

BE_Loop:

; ============================================================================
; Confront - show one piece of evidence to one suspect.
;   The reply comes from a 2D table:  cell = suspect * NUM_CLUES + clue
;   An empty cell (0) means "no reaction".
; ============================================================================
Confront PROC
    push eax
    push ebx
    push ecx
    push edx
    push esi
    push edi

    ; ---- step 1: choose the suspect ----
    call NewScreen
    mov  edx, OFFSET msgPromptConf1
    call WriteString
    mov  esi, OFFSET suspectNameTable
    mov  ecx, NUM_SUSPECTS + 1
    mov  ebx, MENU_ROW
    call ArrowMenu
    cmp  eax, NUM_SUSPECTS
    jae  CF_Done
    mov  edi, eax                ; EDI = suspect index

    ; ---- step 2: choose the evidence ----
    call BuildEvidenceMenu
    call NewScreen
    mov  edx, OFFSET msgPromptConf2
    call WriteString
    mov  esi, OFFSET evMenuTbl
    mov  ecx, NUM_CLUES + 1
    mov  ebx, MENU_ROW
    call ArrowMenu
    cmp  eax, NUM_CLUES
    jae  CF_Done
    mov  ecx, eax                ; ECX = clue index

    cmp  clueFound[ecx], 0
    jne  CF_Have
    mov  edx, OFFSET msgNoEvidence2
    call WriteString
    jmp  CF_Done

CF_Have:
    imul ebx, edi, NUM_CLUES     ; EBX = suspect * NUM_CLUES
    add  ebx, ecx                ; EBX = cell number in the 2D table
    mov  esi, curConfrontTable
    mov  edx, [esi+ebx*4]        ; reply text, or 0 = no reaction
    cmp  edx, 0
    jne  CF_React

    ; ---- no reaction ----
    mov  edx, OFFSET msgNoReact1
    call WriteString
    mov  edx, suspectNameTable[edi*4]
    call WriteString
    mov  edx, OFFSET msgNoReact2
    call WriteString
    jmp  CF_Done

CF_React:
    push edx                     ; keep the reply address
    mov  edx, OFFSET msgConfHeader
    call WriteString
    movzx eax, suspectColor[edi]
    call SetTextColor
    pop  edx
    call TypeString
    call Crlf
    mov  eax, white + (black * 16)
    call SetTextColor

    cmp  confrontDone[ebx], 0
    jne  CF_Old
    mov  confrontDone[ebx], 1
    add  score, POINTS_PROOF
    mov  edx, OFFSET msgNewProof
    call WriteString
    jmp  CF_Done
CF_Old:
    mov  edx, OFFSET msgOldProof
    call WriteString

CF_Done:
    pop  edi
    pop  esi
    pop  edx
    pop  ecx
    pop  ebx
    pop  eax
    ret
Confront ENDP

; ============================================================================
; ToLowerStr - converts a null-terminated string to lower case in place
;   IN : ESI = string address
; ============================================================================
ToLowerStr PROC
    push esi
    push eax
TL_Loop:
    mov  al, [esi]
    cmp  al, 0
    je   TL_Done
    cmp  al, 'A'
    jb   TL_Next
    cmp  al, 'Z'
    ja   TL_Next
    add  al, 32                  ; 'A'..'Z' -> 'a'..'z'
    mov  [esi], al
TL_Next:
    inc  esi
    jmp  TL_Loop
TL_Done:
    pop  eax
    pop  esi
    ret
ToLowerStr ENDP

; ============================================================================
; TrimTrail - removes trailing spaces from a string
;   IN : ESI = string address, EAX = its length (as returned by ReadString)
; ============================================================================
TrimTrail PROC
    push eax
TT_Loop:
    cmp  eax, 0
    je   TT_Done
    cmp  BYTE PTR [esi+eax-1], ' '
    jne  TT_Done
    dec  eax
    mov  BYTE PTR [esi+eax], 0   ; cut the string one character earlier
    jmp  TT_Loop
TT_Done:
    pop  eax
    ret
TrimTrail ENDP

; ============================================================================
; CompareStrings - compares two null-terminated strings
;   IN : ESI = string 1, EDI = string 2
;   OUT: ZF = 1 if equal, ZF = 0 if different
; ============================================================================
CompareStrings PROC
    push esi
    push edi
    push eax
CS_Loop:
    mov  al, [esi]
    mov  ah, [edi]
    cmp  al, ah
    jne  CS_Done                 ; different  -> ZF = 0
    cmp  al, 0
    je   CS_Done                 ; both ended -> ZF = 1
    inc  esi
    inc  edi
    jmp  CS_Loop
CS_Done:
    pop  eax                     ; POP does not change flags
    pop  edi
    pop  esi
    ret
CompareStrings ENDP

; ============================================================================
; SolvePuzzle - puzzle 1 (number answer), puzzle 2 (text answer)
; ============================================================================
SolvePuzzle PROC
    push eax
    push ebx
    push ecx
    push edx
    push esi
    push edi

    call NewScreen
    mov  edx, OFFSET msgPromptPuz
    call WriteString
    mov  esi, OFFSET puzzleMenuTbl
    mov  ecx, NUM_PUZZLES + 1
    mov  ebx, MENU_ROW
    call ArrowMenu
    cmp  eax, NUM_PUZZLES
    jae  SP_Done

    mov  ebx, eax                ; zero-based puzzle index
    cmp  puzzleDone[ebx], 0
    jne  SP_AlreadyDone

    cmp  ebx, 0
    je   SP_Puzzle1
    jmp  SP_Puzzle2

SP_Puzzle1:
    mov  edx, OFFSET puz1Text
    call WriteString
    call ReadInt
    jo   SP_Wrong                ; typed something that is not a number
    cmp  eax, PUZ1_ANSWER
    je   SP_Correct
    jmp  SP_Wrong

SP_Puzzle2:
    mov  edx, OFFSET puz2Text
    call WriteString
    mov  edx, OFFSET inputBuf
    mov  ecx, 30                 ; max characters to read
    call ReadString              ; EAX = number of characters typed
    mov  esi, OFFSET inputBuf
    call TrimTrail               ; "clock  " -> "clock"
    call ToLowerStr              ; accept "CLOCK", "Clock", ...
    mov  esi, OFFSET inputBuf
    mov  edi, OFFSET puz2Answer
    call CompareStrings          ; ZF = 1 if equal
    je   SP_Correct
    jmp  SP_Wrong

SP_Correct:
    mov  puzzleDone[ebx], 1
    add  score, POINTS_PUZZLE
    mov  edx, OFFSET msgPuzRight
    call WriteString
    jmp  SP_Done

SP_Wrong:
    mov  eax, score
    sub  eax, PENALTY_PUZZLE
    jns  SP_StoreScore           ; result not negative -> keep it
    mov  eax, 0                  ; never let the score go below zero
SP_StoreScore:
    mov  score, eax
    mov  edx, OFFSET msgPuzWrong
    call WriteString
    mov  al, 7                   ; bell character = beep
    call WriteChar
    jmp  SP_Done

SP_AlreadyDone:
    mov  edx, OFFSET msgPuzDone
    call WriteString

SP_Done:
    pop  edi
    pop  esi
    pop  edx
    pop  ecx
    pop  ebx
    pop  eax
    ret
SolvePuzzle ENDP

; ============================================================================
; ShowEvidence - clues found, suspect statements, and confrontation replies
; ============================================================================
ShowEvidence PROC
    push eax
    push ebx
    push ecx
    push edx
    push esi
    push edi

    call NewScreen

    ; ---- found clues ----
    mov  edx, OFFSET msgEvidenceHdr
    call WriteString
    mov  edi, curClueTable
    mov  esi, 0                  ; index
    mov  ebx, 0                  ; how many printed
SE_ClueLoop:
    cmp  clueFound[esi], 0
    je   SE_ClueSkip
    inc  ebx
    mov  edx, curEvidTable
    mov  edx, [edx+esi*4]        ; short evidence name
    call WriteString
    mov  al, ':'
    call WriteChar
    call Crlf
    mov  edx, [edi+esi*4]        ; full clue text
    call WriteString
    call Crlf
    call Crlf
SE_ClueSkip:
    inc  esi
    cmp  esi, NUM_CLUES
    jb   SE_ClueLoop
    cmp  ebx, 0
    jne  SE_Notes
    mov  edx, OFFSET msgNoEvidence
    call WriteString

    ; ---- suspect statements ----
SE_Notes:
    mov  edx, OFFSET msgNotesHdr
    call WriteString
    mov  esi, 0
    mov  ebx, 0
SE_SusLoop:
    cmp  suspectAsked[esi], 0
    je   SE_SusSkip
    inc  ebx
    mov  edx, suspectNameTable[esi*4]
    call WriteString
    mov  al, ':'
    call WriteChar
    call Crlf
    mov  edx, suspectStmtTable[esi*4]
    call WriteString
    call Crlf
    call Crlf
SE_SusSkip:
    inc  esi
    cmp  esi, NUM_SUSPECTS
    jb   SE_SusLoop
    cmp  ebx, 0
    jne  SE_Conf
    mov  edx, OFFSET msgNoNotes
    call WriteString

    ; ---- confrontations (nested loop over the 2D table) ----
SE_Conf:
    mov  edx, OFFSET msgConfNotesHdr
    call WriteString
    mov  ebx, 0                  ; how many printed
    mov  esi, 0                  ; suspect
SE_ConfS:
    mov  edi, 0                  ; clue
SE_ConfC:
    imul eax, esi, NUM_CLUES
    add  eax, edi                ; EAX = cell number
    cmp  confrontDone[eax], 0
    je   SE_ConfNext
    inc  ebx
    push eax                     ; remember the cell
    mov  edx, suspectNameTable[esi*4]
    call WriteString
    mov  edx, OFFSET msgVs
    call WriteString
    mov  edx, curEvidTable
    mov  edx, [edx+edi*4]
    call WriteString
    call Crlf
    pop  eax
    mov  edx, curConfrontTable
    mov  edx, [edx+eax*4]
    call WriteString
    call Crlf
    call Crlf
SE_ConfNext:
    inc  edi
    cmp  edi, NUM_CLUES
    jb   SE_ConfC
    inc  esi
    cmp  esi, NUM_SUSPECTS
    jb   SE_ConfS
    cmp  ebx, 0
    jne  SE_Done
    mov  edx, OFFSET msgNoConf
    call WriteString

SE_Done:
    pop  edi
    pop  esi
    pop  edx
    pop  ecx
    pop  ebx
    pop  eax
    ret
ShowEvidence ENDP

    mov  edx, OFFSET msgStatPuz

    pop  edx
    pop  ecx
    pop  eax

ShowStatus ENDP

; ============================================================================
; CheckAccusation - final accusation and ending
;   OUT: EAX = 1 if the case ended, 0 if the player has to keep investigating
; ============================================================================
CheckAccusation PROC
    push ebx
    push ecx
    push edx
    push esi
    push edi

    ; need enough evidence first
    mov  esi, OFFSET clueFound
    mov  ecx, NUM_CLUES
    call CountFlags
    cmp  eax, MIN_CLUES
    jae  CA_Enough
    mov  edx, OFFSET msgNeedMore
    call WriteString
    mov  eax, 0
    jmp  CA_Return

CA_Enough:
    call NewScreen
    mov  edx, OFFSET msgPromptAcc
    call WriteString
    mov  esi, OFFSET suspectNameTable
    mov  ecx, NUM_SUSPECTS + 1
    mov  ebx, MENU_ROW
    call ArrowMenu
    cmp  eax, NUM_SUSPECTS
    jae  CA_Cancel
    mov  edi, eax                ; EDI = accused suspect (zero-based)

    mov  edx, OFFSET msgConfirmQ ; "Are you sure?" pop-up window
    mov  ebx, OFFSET msgConfirmT
    call MsgBoxAsk
    cmp  eax, ANSWER_YES
    jne  CA_Cancel

    cmp  edi, culprit
    jne  CA_Wrong

    ; ---- correct accusation ----
    add  score, POINTS_SOLVED
    mov  eax, lightGreen + (black * 16)
    call SetTextColor
    mov  edx, curSolveText
    call TypeString
    cmp  score, GOOD_SCORE
    jae  CA_Master
    mov  edx, OFFSET msgRankLucky
    jmp  CA_Rank
CA_Master:
    mov  edx, OFFSET msgRankMaster
    jmp  CA_Rank

    ; ---- wrong accusation ----
CA_Wrong:
    mov  eax, lightRed + (black * 16)
    call SetTextColor
    mov  al, 7                   ; beep
    call WriteChar
    mov  edx, OFFSET msgWrong

CA_Rank:
    call WriteString
    call Crlf
    mov  edx, OFFSET msgReveal
    call WriteString
    mov  ebx, culprit
    mov  edx, suspectNameTable[ebx*4]
    call WriteString
    call Crlf
    mov  edx, OFFSET msgFinalScore
    call WriteString
    mov  eax, score
    call WriteDec
    call Crlf
    mov  eax, white + (black * 16)
    call SetTextColor
    mov  eax, 1
    jmp  CA_Return

CA_Cancel:
    mov  eax, 0

CA_Return:
    pop  edi
    pop  esi
    pop  edx
    pop  ecx
    pop  ebx
    ret
CheckAccusation ENDP
